class CustomCard extends StatelessWidget {
  final String text;
  CustomCard({required this.text});

  @override

Custom Widget แบบ Stateful (มี state ของตัวเอง)

class CustomCounterWidget extends StatefulWidget {
  final String title;
  final Color backgroundColor;
  const CustomCounterWidget({super.key, required this.title, required this.backgroundColor});
  @override
  _CustomCounterWidgetState createState() => _CustomCounterWidgetState();
}

class _CustomCounterWidgetState extends State<CustomCounterWidget> {
  int _counter = 0;
  void _incrementCounter() { setState(() { _counter++; }); }

  @override
  Widget build(BuildContext context) {
    return Container(
      padding: const EdgeInsets.all(16),
      decoration: BoxDecoration(color: widget.backgroundColor, borderRadius: BorderRadius.circular(10)),
      child: Column(mainAxisAlignment: MainAxisAlignment.center, children: [
        Text('${widget.title}: $_counter',
             style: const TextStyle(fontSize: 24, fontWeight: FontWeight.bold)),
        const SizedBox(height: 16),
        ElevatedButton(onPressed: _incrementCounter, child: const Text('Increment')),
      ]));
  }
}

เวอร์ชันอัปเกรด: ส่ง callback ขึ้นไปหา parent

เพิ่ม final Function onPlayerWin; แล้วเรียก callback เมื่อ counter ถึง 5 ส่งสีของทีมกลับไปให้ parent เปลี่ยนสีพื้นหลัง

// ใน _CustomCounterWidgetState
void _incrementCounter() {
  setState(() { _counter++; });
  if (_counter == 5) {
    widget.onPlayerWin(widget.backgroundColor);   // เรียก callback ของ parent
  }
}

// ใน parent (StatefulWidget)
Color winPlayerColor = Colors.white;
void setWinPlayerColor(Color winColor) {
  setState(() { winPlayerColor = winColor; });
}
// build: Scaffold(backgroundColor: winPlayerColor, ...)
CustomCounterWidget(title: 'Team A', backgroundColor: Colors.red,  onPlayerWin: setWinPlayerColor),
CustomCounterWidget(title: 'Team B', backgroundColor: Colors.blue, onPlayerWin: setWinPlayerColor),
จำให้ได้: ข้อมูลไหลลงผ่าน constructor (พารามิเตอร์) · เหตุการณ์ไหลขึ้นผ่าน callback · ใน Stateful widget เข้าถึงพารามิเตอร์ด้วย widget.xxx

3. Theme

Theme = การกำหนด style ของ UI ทั้งแอป (สี ฟอนต์ ขนาด รูปแบบ widget) ประโยชน์: UI เป็นระบบเดียวกัน, เปลี่ยนที่เดียวทั้งแอปเปลี่ยน, รองรับ Dark Mode ในตัว

// 1) ประกาศใน MaterialApp
MaterialApp(
  theme: ThemeData(
    primarySwatch: Colors.red,                       // สีหลัก
    textTheme: const TextTheme(
      bodyMedium: TextStyle(fontSize: 18, fontWeight: FontWeight.bold)),
    elevatedButtonTheme: ElevatedButtonThemeData(
      style: ElevatedButton.styleFrom(
        backgroundColor: Colors.deepPurple,          // สีปุ่มหลัก
        foregroundColor: Colors.white)),             // สีข้อความปุ่ม
  ),
  home: const HomeScreen());

// 2) ดึงค่าจาก Theme ใน Widget
Text('Hello World', style: Theme.of(context).textTheme.bodyMedium),
ElevatedButton(onPressed: () {}, child: const Text("Press Me!"))   // ได้สไตล์จาก theme อัตโนมัติ

ใช้ copyWith เพื่อแก้เฉพาะค่าที่ต้องการ จากธีมพื้นฐาน

โจทย์: ปุ่มสลับ Dark/Light บน AppBar (ตัวอย่างวิธีทำ)

Hint จากอาจารย์: ให้ MaterialApp เก็บ ThemeMode แล้วส่งฟังก์ชันสลับกับสถานะไปให้หน้าจอ เพื่อเปลี่ยน icon ปุ่ม (โค้ดด้านล่างผมเขียนเองตาม Hint ไม่ใช่เฉลยของอาจารย์)

class MyApp extends StatefulWidget {
  const MyApp({super.key});
  @override State<MyApp> createState() => _MyAppState();
}
class _MyAppState extends State<MyApp> {
  ThemeMode _mode = ThemeMode.light;
  void _toggle() => setState(() {
    _mode = _mode == ThemeMode.light ? ThemeMode.dark : ThemeMode.light;
  });

  @override
  Widget build(BuildContext context) => MaterialApp(
    theme: ThemeData.light(), darkTheme: ThemeData.dark(), themeMode: _mode,
    home: HomeScreen(mode: _mode, onToggle: _toggle));
}

// ใน HomeScreen: final ThemeMode mode; final VoidCallback onToggle;
AppBar(title: const Text('Profile'), actions: [
  IconButton(
    icon: Icon(mode == ThemeMode.dark ? Icons.light_mode : Icons.dark_mode),
    onPressed: onToggle),
]),

ทำให้ UI มีชีวิตชีวาและเป็นธรรมชาติ ใช้เมื่อสถานะ UI เปลี่ยน เช่น สี ขนาด การเคลื่อนที่ แบ่งเป็น Implicit และ Explicit (ปรับแต่งได้มากกว่าแต่ซับซ้อนกว่า)

6. Implicit Animations

Widget ที่แนะนำ: AnimatedContainer · AnimatedOpacity · AnimatedAlign · AnimatedPadding · AnimatedPositioned · AnimatedSwitcher ทุกตัวมี duration และเปลี่ยนค่าผ่าน setState()

AnimatedContainer

double _size = 100; Color _color = Colors.blue;
void _changeBox() => setState(() {
  _size = _size == 100 ? 200 : 100;
  _color = _color == Colors.blue ? Colors.red : Colors.blue;
});
AnimatedContainer(
  duration: const Duration(seconds: 1),
  curve: Curves.easeInOut,       // รูปแบบการเคลื่อนไหว
  width: _size, height: _size, color: _color,
)

AnimatedOpacity

double _opacity = 1.0;
void _toggleOpacity() => setState(() { _opacity = _opacity == 1.0 ? 0.2 : 1.0; });
AnimatedOpacity(
  duration: const Duration(seconds: 1),
  opacity: _opacity,
  child: Container(width: 100, height: 100, color: Colors.blue),
)

AnimatedAlign

bool _isLeft = true;
void _togglePosition() => setState(() { _isLeft = !_isLeft; });
Stack(children: [
  AnimatedAlign(
    duration: const Duration(seconds: 1),
    alignment: _isLeft ? Alignment.centerLeft : Alignment.centerRight,
    child: Container(width: 100, height: 100, color: Colors.red)),
  Center(child: ElevatedButton(onPressed: _togglePosition, child: const Text('Move Box'))),
])

AnimatedPadding

double _padding = 10;
void _increasePadding() => setState(() { _padding = _padding == 10 ? 50 : 10; });
AnimatedPadding(
  duration: const Duration(seconds: 1),
  padding: EdgeInsets.all(_padding),
  child: Container(width: 100, height: 100, color: Colors.green),
)

AnimatedPositioned (ต้องอยู่ใน Stack)

bool _isLeft = true;
void _moveBox() => setState(() { _isLeft = !_isLeft; });
Stack(children: [
  AnimatedPositioned(
    duration: const Duration(seconds: 1),
    left: _isLeft ? 20 : 200, top: 100,
    child: Container(width: 100, height: 100, color: Colors.purple)),
  Center(child: ElevatedButton(onPressed: _moveBox, child: const Text('Move Box'))),
])

AnimatedSwitcher (สลับ widget ต้องมี key ต่างกัน)

bool _isFirst = true;
void _switchWidget() => setState(() { _isFirst = !_isFirst; });
AnimatedSwitcher(
  duration: const Duration(seconds: 1),
  child: _isFirst
    ? Container(key: const ValueKey(1), width: 100, height: 100, color: Colors.orange)
    : Container(key: const ValueKey(2), width: 100, height: 100, color: Colors.blue),
)
ระวัง: AnimatedPositioned ใช้ใน Stack เท่านั้น · AnimatedSwitcher ต้องให้ child แต่ละตัวมี key ต่างกัน ไม่งั้นไม่เกิด animation

7. Explicit Animation (Advanced)

ผู้พัฒนาควบคุมเองทั้งหมด: เริ่มเมื่อใด หยุดเมื่อใด ความเร็ว และรูปแบบการเปลี่ยนค่า เครื่องมือหลัก: AnimationController, Animation, Tween · Widget: FadeTransition, ScaleTransition, RotationTransition, SlideTransition

ตัวอย่างด้านล่างผมเขียนเองให้เห็นโครงสร้างมาตรฐาน (ในสไลด์เป็นภาพ ผมดึงโค้ดออกมาไม่ได้)

class FadeDemo extends StatefulWidget {
  const FadeDemo({super.key});
  @override State<FadeDemo> createState() => _FadeDemoState();
}
class _FadeDemoState extends State<FadeDemo> with SingleTickerProviderStateMixin {
  late final AnimationController _controller = AnimationController(
    vsync: this, duration: const Duration(seconds: 2));
  late final Animation<double> _fade =
    Tween<double>(begin: 0, end: 1).animate(_controller);

  @override void dispose() { _controller.dispose(); super.dispose(); }  // ต้อง dispose

  @override
  Widget build(BuildContext context) => Column(children: [
    FadeTransition(opacity: _fade, child: const FlutterLogo(size: 100)),
    ElevatedButton(onPressed: () => _controller.forward(), child: const Text('Start')),
    ElevatedButton(onPressed: () => _controller.reverse(), child: const Text('Reverse')),
  ]);
}
จำ: ต้อง with SingleTickerProviderStateMixin เพื่อให้มี vsync: this · สั่งงานด้วย forward(), reverse(), repeat(), stop() · อย่าลืม dispose()

8. แบบฝึกหัด: ไฟจราจร

โจทย์: ไฟแดง/เหลือง/เขียว + ปุ่ม "เปลี่ยนไฟ" วนลูป เริ่มแดง (opacity 1.0) ไฟอื่น 0.3 → แดง→เหลือง→เขียว→กลับแดง Hint: AnimatedOpacity + StatefulWidget (โค้ดผมเขียนเอง)

class TrafficLight extends StatefulWidget {
  const TrafficLight({super.key});
  @override State<TrafficLight> createState() => _TrafficLightState();
}
class _TrafficLightState extends State<TrafficLight> {
  int _current = 0;   // 0=แดง 1=เหลือง 2=เขียว
  void _next() => setState(() { _current = (_current + 1) % 3; });

  Widget _light(Color color, int index) => AnimatedOpacity(
    duration: const Duration(milliseconds: 500),
    opacity: _current == index ? 1.0 : 0.3,
    child: Container(width: 80, height: 80,
      margin: const EdgeInsets.all(8),
      decoration: BoxDecoration(color: color, shape: BoxShape.circle)),
  );

  @override
  Widget build(BuildContext context) => Scaffold(
    appBar: AppBar(title: const Text('Traffic Light')),
    body: Center(child: Column(mainAxisAlignment: MainAxisAlignment.center, children: [
      _light(Colors.red, 0), _light(Colors.yellow, 1), _light(Colors.green, 2),
      const SizedBox(height: 24),
      ElevatedButton(onPressed: _next, child: const Text('เปลี่ยนไฟ')),
    ])),
  );
}

หัวใจคือ (_current + 1) % 3 ทำให้วนกลับไปแดงหลังเขียว

9. ข้อสอบจำลอง (กดดูเฉลย)

1) UI Composition คืออะไร และมีข้อดีอย่างไร?

การประกอบ Widget เล็ก ๆ เป็นหน้าจอใหญ่ ข้อดี: อ่านง่าย โครงสร้างชัด แก้ง่าย นำกลับมาใช้ซ้ำได้

2) แนวคิดสำคัญของ Flutter ที่เกี่ยวกับ UI คืออะไร?

Everything is a Widget

3) ใน Stateful widget เข้าถึง property ของ widget ด้วยอะไร?

widget.xxx เช่น widget.title

4) ส่งข้อมูลจาก child กลับไป parent ทำอย่างไร?

ส่งฟังก์ชัน (callback) เข้าไปทาง constructor แล้ว child เรียกใช้ เช่น onPlayerWin

5) ธีมมืดกำหนดที่พารามิเตอร์ใดของ MaterialApp และสลับด้วยอะไร?

darkTheme สลับด้วย themeMode (system/light/dark)

6) ข้อแตกต่างระหว่าง Implicit กับ Explicit Animation?

Implicit: ง่าย ใช้ setState ไม่ต้องมี Controller เหมาะ UI ทั่วไป · Explicit: ใช้ AnimationController คุมเองทั้งหมด เหมาะ animation ซับซ้อน

7) ข้อใดไม่ใช่ Implicit Animation: AnimatedOpacity, AnimatedAlign, FadeTransition, AnimatedPadding

FadeTransition (เป็น Explicit ต้องใช้ Animation/Controller)

8) AnimatedPositioned ต้องอยู่ใน widget ใด?

Stack

9) (อัตนัย) อธิบายวิธีทำไฟจราจรด้วย AnimatedOpacity

ใช้ StatefulWidget เก็บ index ไฟปัจจุบัน แต่ละดวงครอบด้วย AnimatedOpacity: opacity = 1.0 ถ้าเป็นดวงปัจจุบัน ไม่งั้น 0.3 · ปุ่มเรียก setState เพิ่ม index แบบ (i+1) % 3
